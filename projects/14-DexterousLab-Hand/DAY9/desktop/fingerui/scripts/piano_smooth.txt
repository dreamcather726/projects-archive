# 流畅弹钢琴脚本 - 手指动作连贯
# 手指对应: 1=拇指, 2=食指, 3=中指, 4=无名指
# 90度=抬起, 180度=按下

# 初始状态 - 所有手指抬起
ALL:90,90,90,90
DELAY:300

# ========== 流畅音阶练习 ==========
# C-D-E-F-E-D-C - 手指连贯动作

# 拇指按下 C
ALL:180,90,90,90
DELAY:200
# 拇指抬起同时食指按下 D
ALL:90,180,90,90
DELAY:200
# 食指抬起同时中指按下 E
ALL:90,90,180,90
DELAY:200
# 中指抬起同时无名指按下 F
ALL:90,90,90,180
DELAY:200
# 无名指抬起同时中指按下 E
ALL:90,90,180,90
DELAY:200
# 中指抬起同时食指按下 D
ALL:90,180,90,90
DELAY:200
# 食指抬起同时拇指按下 C
ALL:180,90,90,90
DELAY:400
# 拇指抬起
ALL:90,90,90,90
DELAY:300

# ========== 流畅和弦练习 ==========
# C大三和弦 (C-E) - 同时按下
ALL:180,90,180,90
DELAY:400
# 同时抬起
ALL:90,90,90,90
DELAY:200

# G大三和弦 (D-G) - 同时按下
ALL:90,180,90,180
DELAY:400
# 同时抬起
ALL:90,90,90,90
DELAY:200

# C

DELAY:150

# G (保持)

# 中指抬起同时无名指按下 A
ALL:90,90,90,180
DELAY:150
# A (保持)
ALL:90,90,90,180
DELAY:150
# 无名指抬起同时中指按下 G (长音)
ALL:90,90,180,90
DELAY:300
# 中指抬起
ALL:90,90,90,90
DELAY:300

# ========== 琶音练习 - 流畅版 ==========
# C-E-G-C 快速连续

# 拇指按下 C

DELAY:80
# 拇指抬起同时中指按下 E
ALL:90,90,180,90
DELAY:80
# 中指抬起同时无名指按下 G

DELAY:80

DELAY:80
# 中指抬起同时拇指按下 C
ALL:180,90,90,90
DELAY:80

ALL:90,90,90,90
DELAY:300

# ========== 快速轮指练习 ==========
# 1-2-3-4-3-2-1 快速连贯

ALL:180,90,90,90
DELAY:60
# 拇指抬起同时食指按下
ALL:90,180,90,90
DELAY:60

# ========== 和弦转位练习 ==========
# C和弦 -> G和弦 -> C和弦 快速转换

DELAY:100
